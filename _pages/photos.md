<!--
  Photo gallery, benchmarked on https://junyonglee.me/photos/ (justified rows + full-screen viewer).
  The photo list lives in _data/photos.yml; image files go in assets/img/photos/.
  Thumbnails use the WebP sizes generated by jekyll-imagemagick (only those smaller than the
  original), and the viewer always loads the untouched original file.
-->

<link rel="stylesheet" href="{{ site.third_party_libraries.photoswipe.url.css }}">

<div
  class="photo-gallery"
  id="photo-gallery"
  data-pswp-lightbox="{{ site.third_party_libraries.photoswipe-lightbox.url.js }}"
  data-pswp-core="{{ site.third_party_libraries.photoswipe.url.js }}"
>
  {% for photo in site.data.photos %}
    {% assign src = 'assets/img/photos/' | append: photo.file %}
    {% assign stem = src | remove: '.jpg' | remove: '.jpeg' | remove: '.png' %}
    {% assign ratio = photo.width | times: 1.0 | divided_by: photo.height %}
    <a
      class="photo-item"
      href="{{ src | relative_url }}"
      data-pswp-width="{{ photo.width }}"
      data-pswp-height="{{ photo.height }}"
      data-caption="{{ photo.caption | escape }}"
      style="--ar: {{ ratio }}"
    >
      <img
        src="{{ src | relative_url }}"
        srcset="{% if site.imagemagick.enabled %}{% for w in site.imagemagick.widths %}{% if w < photo.width %}{{ stem | relative_url }}-{{ w }}.webp {{ w }}w, {% endif %}{% endfor %}{% endif %}{{ src | relative_url }} {{ photo.width }}w"
        sizes="(max-width: 576px) {{ ratio | times: 180 | round }}px, {{ ratio | times: 420 | round }}px"
        width="{{ photo.width }}"
        height="{{ photo.height }}"
        alt="{{ photo.caption | default: 'Photo by Junseo Bang' | escape }}"
        loading="{% if forloop.index <= 8 %}eager{% else %}lazy{% endif %}"
        decoding="async"
      >
    </a>
  {% endfor %}
</div>
